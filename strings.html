#include <ctype.h>

int main() {
    char str[100];
    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    int capNext = 1;
    for(int i=0; str[i]!='\0'; i++){
        if(isspace(str[i]))
            capNext = 1;
        else if(capNext){
            str[i] = toupper(str[i]);
            capNext = 0;
        } else {
            str[i] = tolower(str[i]);
        }
    }

    printf("\nString in title case: %s\n", str);
    return 0;
}

Output: Enter a string: hello world from C

String in title case: Hello World From C

int main() {
    char str1[100], str2[100];
    printf("Enter first string: ");
    fgets(str1, sizeof(str1), stdin);
    printf("Enter second string: ");
    fgets(str2, sizeof(str2), stdin);

    // Remove newline characters
    str1[strcspn(str1, "\n")] = 0;
    str2[strcspn(str2, "\n")] = 0;

    int cmp = strcmp(str1, str2);
    if(cmp == 0)
        printf("\nStrings are equal.\n");
    else
        printf("\nStrings are not equal.\n");

    return 0;
}

#include <stdio.h>
#include <string.h>

int main() {
    char str[100], rev[100];
    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    int len = strlen(str);
    if(str[len-1] == '\n') len--;

    for(int i=0; i<len; i++) {
        rev[i] = str[len-1-i];
    }
    rev[len] = '\0';

    printf("\nReversed string: %s\n", rev);
    return 0;
}

Reversed string: olleH

    return 0;

#include <stdio.h>
#include <string.h>

int main() {
    char str[100], rev[100];
    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    int len = strlen(str);
    if(str[len-1]=='\n') len--;

    for(int i=0; i<len; i++)
        rev[i] = str[len-1-i];
    rev[len] = '\0';

    if(strncmp(str, rev, len) == 0)
        printf("\nString is a palindrome.\n");
    else
        printf("\nString is not a palindrome.\n");

    return 0;
}

#include <stdio.h>

int main() {
    int roll;
    char name[50], className[20];
    int marks[5];

    printf("Enter roll number: ");
    scanf("%d", &roll);
    getchar(); // consume newline

    printf("Enter name: ");
    fgets(name, sizeof(name), stdin);
    name[strcspn(name,"\n")] = 0;

    printf("Enter class: ");
    fgets(className, sizeof(className), stdin);
    className[strcspn(className,"\n")] = 0;

    for(int i=0; i<5; i++){
        printf("Enter marks of subject %d: ", i+1);
        scanf("%d", &marks[i]);
    }

    printf("\nStudent Info:\nRoll No: %d\nName: %s\nClass: %s\nMarks: ", roll, name, className);
    for(int i=0; i<5; i++) printf("%d ", marks[i]);
    printf("\n");

    return 0;
}
