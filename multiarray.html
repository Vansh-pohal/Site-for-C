#include <stdio.h>
int main(){
    int arr[2][2], rev[2][2];
    printf("Enter elements for a 2x2 matrix:\n");
    for(int i=0;i<2;i++)
        for(int j=0;j<2;j++){
            printf("Enter element at row %d, column %d: ", i+1, j+1);
            scanf("%d", &arr[i][j]);
        }

    for(int i=0;i<2;i++)
        for(int j=0;j<2;j++)
            rev[i][j] = arr[1-i][1-j];

    printf("\nArray copied in reverse order:\n");
    for(int i=0;i<2;i++){
        for(int j=0;j<2;j++)
            printf("%d\t", rev[i][j]);
        printf("\n");
    }
    return 0;
}

Enter element at row 1, column 1: 5
Enter element at row 1, column 2: 3
Enter element at row 2, column 1: -2
Enter element at row 2, column 2: 7

int main(){
    int arr[2][2];

#include <stdio.h>
int main(){
    int arr[5], temp;
    printf("Enter 5 values of the array:\n");
    for(int i=0;i<5;i++){
        printf("Enter value %d: ", i+1);
        scanf("%d", &arr[i]);
    }

    // Bubble sort ascending
    for(int i=0;i<5-1;i++)
        for(int j=0;j<5-i-1;j++)
            if(arr[j]>arr[j+1]){
                temp=arr[j];
                arr[j]=arr[j+1];
                arr[j+1]=temp;
            }

    printf("\nArray in ascending order:\n");
    for(int i=0;i<5;i++) printf("%d ", arr[i]);
    printf("\n");
    return 0;
}

Output: Enter 5 values of the array:
Enter value 1: 5
Enter value 2: 2
Enter value 3: 8
Enter value 4: 1
Enter value 5: 4

Array in ascending order:
1 2 4 5 8
