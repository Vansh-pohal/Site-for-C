  printf("Enter a number: ");
  scanf("%d",&n);
  absn = (n<0)? -n : n;
  if(absn <= 9)
    printf("The entered number is 1 digit.");
  else if(absn <= 99)
    printf("The entered number is 2 digit.");
  else if(absn <= 999)
    printf("The entered number is 3 digit.");
  else if(absn <= 9999)
    printf("The entered number is 4 digit.");
  else if(absn <= 99999)
    printf("The entered number is 5 digit.");

    printf("The entered number has more than 5 digits.");
  return 0;
}

  printf("Enter date (dd mm yyyy): ");
  scanf("%d%d%d",&d,&m,&y);
  if(m < 1 || m > 12){
    printf("Invalid date (month out of range).");
  } else if(d < 1 || d > 31){
    printf("Invalid date (day out of range).");
  } else if(m==2){
    if( (y%400==0) || (y%4==0 && y%100!=0) ){
      if(d > 29) printf("Invalid date for February in leap year.");
      else printf("Valid date.");
    } else {
      if(d > 28) printf("Invalid date for February.");
      else printf("Valid date.");
    }
  } else if(m==4 || m==6 || m==9 || m==11){
    if(d > 30) printf("Invalid date for given month.");
    else printf("Valid date.");
  } else {
    printf("Valid date.");
  }
  return 0;
}

#include <stdio.h>
int main(){
  int bd, bm, by;
  int cd, cm, cy;
  printf("Enter birth date (dd mm yyyy): ");
  scanf("%d%d%d",&bd,&bm,&by);
  printf("Enter current date (dd mm yyyy): ");
  scanf("%d%d%d",&cd,&cm,&cy);
  int ageYears = cy - by;
  int ageMonths = cm - bm;
  int ageDays = cd - bd;
  if(ageDays < 0){
    ageDays += 30; /* simple borrowing, not exact calendar math */
    ageMonths--;
  }
  if(ageMonths < 0){
    ageMonths += 12;
    ageYears--;
  }
  printf("Age: %d years, %d months, %d days", ageYears, ageMonths, ageDays);
  return 0;
}

Output: Enter birth date (dd mm yyyy) : 15 08 2000
Enter current date (dd mm yyyy) : 05 11 2025
Age: 25 years, 2 months, 21 days
