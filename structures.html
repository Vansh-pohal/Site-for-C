STRUCTURES

Q1.Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ?

Q2. Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print only name of the employee ?

Q3. Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form ?

Q4. Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?

CLOSE 
#include <stdio.h>

struct Employee {
    int id;
    char name[50];
    char designation[50];
};

int main() {
    struct Employee emp[50];
    int n;
    printf("Enter number of employees: ");
    scanf("%d", &n);
    getchar(); // consume newline

    for(int i=0; i<n; i++){
        printf("\nEmployee %d ID: ", i+1);
        scanf("%d", &emp[i].id);
        getchar();
        printf("Employee %d Name: ", i+1);
        fgets(emp[i].name, sizeof(emp[i].name), stdin);
        emp[i].name[strcspn(emp[i].name, "\n")] = 0;
        printf("Employee %d Designation: ", i+1);
        fgets(emp[i].designation, sizeof(emp[i].designation), stdin);
        emp[i].designation[strcspn(emp[i].designation, "\n")] = 0;
    }

    printf("\nID\tName\t\tDesignation\n");
    printf("----------------------------------------\n");
    for(int i=0; i<n; i++){
        printf("%d\t%s\t\t%s\n", emp[i].id, emp[i].name, emp[i].designation);
    }

    return 0;
}

#include <stdio.h>
#include <string.h>

struct Student {
    int roll;
    char name[50];
    float fee;
    char dob[15];
};

int main() {
    struct Student s[50];
    int n;
    printf("Enter number of students: ");
    scanf("%d", &n);
    getchar();

    for(int i=0; i<n; i++){
        printf("\nStudent %d Roll: ", i+1);
        scanf("%d", &s[i].roll);
        getchar();
        printf("Student %d Name: ", i+1);
        fgets(s[i].name, sizeof(s[i].name), stdin);
        s[i].name[strcspn(s[i].name, "\n")] = 0;
        printf("Student %d Fee: ", i+1);
        scanf("%f", &s[i].fee);
        getchar();
        printf("Student %d DOB: ", i+1);
        fgets(s[i].dob, sizeof(s[i].dob), stdin);
        s[i].dob[strcspn(s[i].dob, "\n")] = 0;
    }

    // Sort by roll number
    struct Student temp;
    for(int i=0;i<n-1;i++){
        for(int j=i+1;j<n;j++){
            if(s[i].roll > s[j].roll){
                temp = s[i];
                s[i] = s[j];
                s[j] = temp;
            }
        }
    }

    printf("\nRoll\tName\t\tFee\tDOB\n");
    printf("---------------------------------------------\n");
    for(int i=0;i<n;i++){
        printf("%d\t%s\t\t%.2f\t%s\n", s[i].roll, s[i].name, s[i].fee, s[i].dob);
    }

    return 0;
}
