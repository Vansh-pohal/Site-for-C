#include <stdio.h>

void main()
{
    int a = 5, b;

    b = ++a;   // pre-increment: a is increased first, then assigned to b

    printf("Value of a = %d\n", a);
    printf("Value of b = %d\n", b);
}

Output: Value of a = 6
Value of b = 6

          #include <stdio.h>

void main()
{
    int a, b, sum, sub, mul, rem;
    float div;

    // Addition
    printf("Enter two numbers for addition: ");
    scanf("%d%d", &a, &b);
    sum = a + b;
    printf("Sum = %d\n\n", sum);

    // Subtraction
    printf("Enter two numbers for subtraction: ");
    scanf("%d%d", &a, &b);
    sub = a - b;
    printf("Subtraction = %d\n\n", sub);

    // Multiplication
    printf("Enter two numbers for multiplication: ");
    scanf("%d%d", &a, &b);
    mul = a * b;
    printf("Multiplication = %d\n\n", mul);

    // Division
    printf("Enter two numbers for division: ");
    scanf("%d%d", &a, &b);
    div = (float)a / b;  // typecasting for accurate float result
    printf("Division = %.2f\n\n", div);

    // Remainder
    printf("Enter two numbers to find remainder: ");
    scanf("%d%d", &a, &b);
    rem = a % b;
    printf("Remainder = %d\n", rem);
}

Output:
Enter two numbers for addition: 5 3
Sum = 8

Enter two numbers for subtraction: 9 4
Subtraction = 5

Enter two numbers for multiplication: 6 7
Multiplication = 42

Enter two numbers for division: 10 2
Division = 5.00

Enter two numbers to find remainder: 10 3
Remainder = 1

#include <stdio.h>

void main()
{
    float num;
    int intPart, rightPart;

    printf("Enter any decimal number: ");
    scanf("%f", &num);

    intPart = (int)num;             // take integer part
    rightPart = (num - intPart) * 10000;  // multiply to shift decimals (4 digits)

    printf("Digits after decimal point: %d", rightPart);
}

Output: Enter any decimal number: 45.6789
Digits after decimal point: 6789
