Q1. Enter an array from user and display all the values?

    for(int i=0;i<n;i++){
        printf("Enter element %d: ", i+1); scanf("%d", &arr[i]);
    }

Q3. Enter an array from user and copy the array into second array?

CLOSE 
#include <stdio.h>
int main(){
    int n;
    printf("Enter the size of the array: ");
    scanf("%d", &n);

    int arr1[n], arr2[n];

    // Input first array
    for(int i=0; i<n; i++){
        printf("Enter element %d: ", i+1);
        scanf("%d", &arr1[i]);
    }

    // Copy to second array
    for(int i=0; i<n; i++){
        arr2[i] = arr1[i];
    }

    // Print second array
    printf("Second array (copied): ");
    for(int i=0; i<n; i++){
        printf("%d ", arr2[i]);
    }

    return 0;
}

Output: Enter the size of the array: 5
Enter element 1: 10
Enter element 2: 20
Enter element 3: 30
Enter element 4: 40
Enter element 5: 50
Second array (copied): 10 20 30 40 50

Q4. Enter an array from user and copy it into another array in reverse order?

CLOSE 
#include <stdio.h>

int main() {
    int n;
    printf("Enter the size of the array: ");
    scanf("%d", &n);

    int arr[n], rev[n];

    // Input array
    for(int i = 0; i < n; i++) {
        printf("Enter element %d: ", i+1);
        scanf("%d", &arr[i]);
    }

    // Reverse copy
    for(int i = 0; i < n; i++) {
        rev[i] = arr[n-1-i];
    }

    // Print reversed array
    printf("Reversed copy: ");
    for(int i = 0; i < n; i++) {
        printf("%d ", rev[i]);
    }

    return 0;
}

Output: Enter the size of the array: 4
Enter element 1: 1
Enter element 2: 2
Enter element 3: 3
Enter element 4: 4
Reversed copy: 4 3 2 1

Q11. Enter five values of array and sort in ascending order?

CLOSE 
#include <stdio.h>
int main(){
    int arr[5], temp;
    for(int i=0;i<5;i++) scanf("%d", &arr[i]);
    for(int i=0;i<5-1;i++){
        for(int j=i+1;j<5;j++){
            if(arr[i]>arr[j]){
                temp=arr[i]; arr[i]=arr[j]; arr[j]=temp;
            }
        }
    }
    printf("Sorted array: ");
    for(int i=0;i<5;i++) printf("%d ", arr[i]);
    return 0;
}

Output: 1 2 3 4 5

Q12. Find maximum value of an array?

CLOSE 
#include <stdio.h>
int main(){
    int n; scanf("%d", &n);
    int arr[n];
    for(int i=0;i<n;i++) scanf("%d", &arr[i]);
    int max = arr[0];
    for(int i=1;i<n;i++){
        if(arr[i]>max) max = arr[i];
    }
    printf("Maximum value: %d", max);
    return 0;
}

Output: 99 (for input 23 45 67 99 12)
