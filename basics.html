           #include <stdio.h>
#include <conio.h>  // for getch()

void main()
{
    char pass[20];
    int i = 0;
    char ch;

    printf("Enter password: ");
    while (1)
    {
        ch = getch();           // reads a character without showing it
        if (ch == 13)           // if Enter is pressed (ASCII 13)
            break;
        pass[i] = ch;
        i++;
        printf("*");            // show * instead of the character
    }
    pass[i] = '\0';             // terminate the string

    printf("\nYour password is: %s", pass); 
}

Output: Enter password: ********
Your password is: mypass

    Notes:
  • ASCII 0-31 are non-printable control characters (so they may look blank or weird).
  • 32-126 are the normal visible ones — like letters, digits, and symbols.
  • 127 is the DEL (delete) character.
